async function searchPapers(q){const url='https://api.openalex.org/works?search='+encodeURIComponent(q)+'&per-page=8&select=id,display_name,publication_year,doi,primary_location,authorships';const r=await fetch(url);if(!r.ok)throw new Error('Research search unavailable');const j=await r.json();return (j.results||[]).map(x=>({title:x.display_name,year:x.publication_year,doi:x.doi,url:x.primary_location?.landing_page_url||x.doi||x.id,authors:(x.authorships||[]).slice(0,3).map(a=>a.author?.display_name).filter(Boolean).join(', ')}))}
function heuristic(papers){if(papers.length>=6)return 78;if(papers.length>=3)return 62;if(papers.length>=1)return 45;return 20}
function feasibility(v){return Math.round((+v.time + +v.sample + +v.data + +v.resources)/16*100)}
async function analyze(e){e.preventDefault();const btn=document.querySelector('#analyze');btn.disabled=true;btn.textContent='กำลังค้นหลักฐานและวิเคราะห์…';const input={idea:idea.value,field:field.value,population:population.value,problem:problem.value,time:time.value,sample:sample.value,data:data.value,resources:resources.value};try{state.papers=await searchPapers(input.idea+' '+input.field);const ev=heuristic(state.papers),feas=feasibility(input);let analysis={summary:'',evidence_assessment:`พบงานวิจัยที่เกี่ยวข้อง ${state.papers.length} รายการ (ผลค้นเป็น evidence candidates)`,potential_gap:'ยังไม่ควรสรุปว่าเป็น Research Gap ที่พิสูจน์แล้ว ต้องตรวจ full text',gap_confidence:'Low',feasibility_assessment:`Feasibility score ${feas}/100`,recommendation:ev>=60&&feas>=75?'GO':ev<40&&feas<40?'STOP':'MODIFY',research_question:'',objective:'',variables:'',methodology:'',next_actions:'ตรวจสอบบทความฉบับเต็มและเปรียบเทียบ population, context, variables, method และ findings'};
